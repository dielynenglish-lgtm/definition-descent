function renderSeat(seatIndex) { const seatEl = document.getElementById(`seat-${seatIndex}`); const customer = gameState.seats[seatIndex]; if (!customer) { seatEl.innerHTML = `
🪑 Empty Seat
`; seatEl.classList.remove('border-rose-500', 'selected-target'); return; } const patiencePct = Math.max(0, (customer.patience / customer.maxPatience) * 100); let barColor = 'bg-emerald-500'; if (patiencePct < 50) barColor = 'bg-amber-500'; if (patiencePct < 25) barColor = 'bg-rose-500 animate-pulse'; // Only render full structure if it's not already rendered for this customer if (!seatEl.querySelector('.speech-bubble')) { seatEl.innerHTML = `
Customer Order: 💬

"${customer.vocab.def}"

${customer.avatar.svg}
${customer.avatar.name}
`; } else { // Smoothly update only the progress bar during game loop ticks const bar = seatEl.querySelector('.patience-bar'); if (bar) { bar.className = `patience-bar ${barColor} h-full transition-all duration-200`; bar.style.width = `${patiencePct}%`; } } }
