    <script>
        const vocabularyQuestions = [
            {
                word: "recipe",
                question: "Fill in the blank: Follow this ______ carefully to make the cake.",
                options: ["recipe", "variation", "combination", "generation"],
                correct: 0
            },
            {
                word: "pasta",
                question: "Fill in the blank: Italian cuisine is famous for spaghetti and other forms of ______.",
                options: ["combination", "pasta", "deserve", "recipe"],
                correct: 1
            },
            {
                word: "variation",
                question: "Fill in the blank: There is a slight ______ in color between the two prints.",
                options: ["generation", "ultimately", "variation", "recipe"],
                correct: 2
            },
            {
                word: "combination",
                question: "Fill in the blank: The lock requires a three-digit secret ______.",
                options: ["combination", "deserve", "pasta", "generation"],
                correct: 0
            },
            {
                word: "generation",
                question: "Fill in the blank: The younger ______ relies heavily on digital technology.",
                options: ["variation", "recipe", "ultimately", "generation"],
                correct: 3
            },
            {
                word: "ultimately",
                question: "Fill in the blank: After hours of debate, they ______ reached an agreement.",
                options: ["deserve", "ultimately", "combination", "pasta"],
                correct: 1
            },
            {
                word: "deserve",
                question: "Fill in the blank: After working so hard all week, you ______ a break.",
                options: ["deserve", "variation", "recipe", "generation"],
                correct: 0
            }
        ];

        const prizeMoney = ["$1,000", "$5,000", "$10,000", "$25,000", "$50,000", "$100,000", "$1,000,000"];

        let currentQuestions = [];
        let currentStep = 0;
        let timeLeft = 60;
        let timerInterval = null;

        function shuffleArray(array) {
            const shuffled = [...array];
            for (let i = shuffled.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
            }
            return shuffled;
        }

        function renderLadder() {
            const ladderEl = document.getElementById('ladder');
            ladderEl.innerHTML = '';
            prizeMoney.forEach((amount, idx) => {
                const stepEl = document.createElement('div');
                stepEl.className = 'ladder-step';
                stepEl.id = `step-${idx}`;
                stepEl.innerText = `${idx + 1}. ${amount}`;
                ladderEl.appendChild(stepEl);
            });
        }

        function updateLadderUI() {
            prizeMoney.forEach((_, idx) => {
                const stepEl = document.getElementById(`step-${idx}`);
                stepEl.className = 'ladder-step';
                if (idx < currentStep) {
                    stepEl.classList.add('completed');
                } else if (idx === currentStep) {
                    stepEl.classList.add('active');
                }
            });
        }

        function startGame() {
            currentQuestions = shuffleArray(vocabularyQuestions).map(q => {
                const originalCorrect = q.options[q.correct];
                const shuffledOptions = shuffleArray(q.options);
                return {
                    question: q.question,
                    options: shuffledOptions,
                    correct: shuffledOptions.indexOf(originalCorrect)
                };
            });

            currentStep = 0;
            timeLeft = 60;

            document.getElementById('overlay').classList.add('hidden');
            document.getElementById('timer').innerText = timeLeft;

            renderLadder();
            updateLadderUI();
            loadQuestion();

            clearInterval(timerInterval);
            timerInterval = setInterval(() => {
                timeLeft--;
                document.getElementById('timer').innerText = timeLeft;
                if (timeLeft <= 0) {
                    endGame(false, "Time's up! You ran out of time.");
                }
            }, 1000);
        }

        function loadQuestion() {
            if (currentStep >= currentQuestions.length) {
                endGame(true, "CONGRATULATIONS! You reached the top of the Money Ladder!");
                return;
            }

            const q = currentQuestions[currentStep];
            document.getElementById('question-text').innerText = q.question;
            
            const optionsContainer = document.getElementById('options-container');
            optionsContainer.innerHTML = '';

            const prefixes = ['A', 'B', 'C', 'D'];
            q.options.forEach((opt, idx) => {
                const btn = document.createElement('button');
                btn.className = 'option-btn';
                btn.innerText = `${prefixes[idx]}: ${opt}`;
                btn.onclick = () => checkAnswer(idx);
                optionsContainer.appendChild(btn);
            });

            updateLadderUI();
        }

        function checkAnswer(selectedIndex) {
            const q = currentQuestions[currentStep];
            if (selectedIndex === q.correct) {
                currentStep++;
                if (currentStep < currentQuestions.length) {
                    loadQuestion();
                } else {
                    endGame(true, "GRAND PRIZE WINNER! You mastered all vocabulary words!");
                }
            } else {
                endGame(false, "Wrong answer! Game Over.");
            }
        }

        function endGame(isWin, message) {
            clearInterval(timerInterval);
            const overlay = document.getElementById('overlay');
            const title = document.getElementById('overlay-title');
            const msg = document.getElementById('overlay-message');
            const btn = document.getElementById('start-btn');

            title.innerText = isWin ? "VICTORY!" : "GAME OVER";
            msg.innerText = message + ` You walked away with: ${currentStep > 0 ? prizeMoney[currentStep - 1] : '$0'}.`;
            btn.innerText = "TRY AGAIN";

            overlay.classList.remove('hidden');
        }

        renderLadder();
    </script>
